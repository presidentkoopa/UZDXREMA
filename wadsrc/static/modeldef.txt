// Rigged VR hand models.
//
// See zscript/vr_hands.zs for the psprite layers these ride on, and why the
// psprite path is used rather than world actors.
//
// NOAUTOREVERSE is the important flag here. GetWeaponTransform mirrors whichever
// model lands on the left physical controller, so that a single weapon mesh can
// serve both hands. These are two purpose-built left and right meshes, so that
// mirror would flip an already-correct hand. Normally the mirror is suppressed by
// +Weapon.NoAutoReverse on the weapon in that slot, but a hand is not a Weapon
// and cannot carry a Weapon flag -- hence the MODELDEF-level flag.
//
// Scale 1.0: the models are exported at 1875x the source FBX, giving a 32.95 unit
// long axis. That matches fist.md3 from VanillaVRPlus (32.89 units at Scale 1.0),
// a hand already working on this same path. Live tuning is vr_hand_scale, which
// rides the psprite scale channel instead of this.

Model RS_VRHandLeft
{
	Path "models/hands"
	Model 0 "hand_left.iqm"
	Skin 0 "hand_basecolor.png"
	Scale 1.0 1.0 1.0
	NOAUTOREVERSE

	// The two source hands sat side by side in the same scene and each kept its
	// own world position, so their origins differ: measured on X, the left
	// hand's geometry centres at -19.55 and the right at +1.40. Left will
	// therefore render well off to the side until this is dialled in.
	//
	// Left at 0 deliberately for the first look. The offset axes here are not
	// model XYZ -- models.cpp applies them as translate(x/xscale, z/zscale,
	// y/yscale), before a further 90 degree yaw -- so guessing a correction
	// risks stacking a second unknown on top of the first. Fix it once we can
	// see which way it actually moves.
	Offset 0.0 0.0 0.0

	// Required, and must come last. BaseFrame is the only thing that inserts
	// into BaseSpriteModelFrames and sets hasmodel, which is what a
	// +DECOUPLEDANIMATIONS actor's model lookup reads. Without it FindModelFrame
	// returns null, the psprite falls through to the sprite path, and the model
	// never draws at all. Last because the insert snapshots the settings above.
	BaseFrame
}

Model RS_VRHandRight
{
	Path "models/hands"
	Model 0 "hand_right.iqm"
	Skin 0 "hand_basecolor.png"
	Scale 1.0 1.0 1.0
	NOAUTOREVERSE
	Offset 0.0 0.0 0.0
	BaseFrame
}
